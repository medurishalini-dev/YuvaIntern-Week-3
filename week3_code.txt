# Dataset: PimaIndiansDiabetes2 from mlbench
# Task: Hypothesis testing + classification + cross-validation + diagnostics

# 1. Install/load packages
packages <- c("mlbench", "tidyverse", "caret", "pROC")
new_packages <- packages[!(packages %in% installed.packages()[,"Package"])]
if(length(new_packages)) install.packages(new_packages)
lapply(packages, library, character.only = TRUE)

set.seed(123)

# 2. Load public dataset
data("PimaIndiansDiabetes2")
diabetes <- PimaIndiansDiabetes2

# 3. Inspect data
dim(diabetes)
str(diabetes)
summary(diabetes)

# 4. Missing-value check
colSums(is.na(diabetes))

# 5. Handle missing values
# This dataset contains missing values represented as NA in several predictors.
# For a classification workflow, median imputation is used for numeric predictors.
diabetes_clean <- diabetes %>%
  mutate(
    across(
      where(is.numeric),
      ~ ifelse(is.na(.x), median(.x, na.rm = TRUE), .x)
    ),
    diabetes = factor(diabetes, levels = c("neg", "pos"))
  )

# 6. Exploratory statistical analysis
# Hypothesis test 1: Is BMI associated with diabetes outcome?
# H0: Mean BMI is the same for negative and positive outcome groups.
# H1: Mean BMI differs between the two groups.
t_test_bmi <- t.test(bmi ~ diabetes, data = diabetes_clean)
t_test_bmi

# Hypothesis test 2: Is glucose level associated with diabetes outcome?
# H0: Mean glucose is the same for negative and positive groups.
# H1: Mean glucose differs between groups.
t_test_glucose <- t.test(glucose ~ diabetes, data = diabetes_clean)
t_test_glucose

# Correlation analysis
cor.test(diabetes_clean$glucose, diabetes_clean$bmi, method = "pearson")
cor.test(diabetes_clean$glucose, diabetes_clean$age, method = "pearson")

# 7. Distribution checks
shapiro.test(diabetes_clean$glucose)
shapiro.test(diabetes_clean$bmi)

# Note: With moderate/large samples, Shapiro-Wilk can be sensitive.
# Histograms and Q-Q plots should also be considered.
ggplot(diabetes_clean, aes(x = glucose)) +
  geom_histogram(bins = 30) +
  labs(title = "Distribution of Glucose", x = "Glucose", y = "Count")

ggplot(diabetes_clean, aes(sample = glucose)) +
  stat_qq() + stat_qq_line() +
  labs(title = "Q-Q Plot of Glucose")

# 8. Train/test split
set.seed(123)
train_index <- createDataPartition(diabetes_clean$diabetes,
                                   p = 0.80, list = FALSE)
train_data <- diabetes_clean[train_index, ]
test_data <- diabetes_clean[-train_index, ]

# 9. Logistic regression model
logit_model <- glm(
  diabetes ~ pregnancies + glucose + pressure + triceps + insulin +
    mass + pedigree + age,
  data = train_data,
  family = binomial
)
summary(logit_model)

# 10. 10-fold cross-validation with caret
set.seed(123)
ctrl <- trainControl(
  method = "cv",
  number = 10,
  classProbs = TRUE,
  summaryFunction = twoClassSummary,
  savePredictions = "final"
)

# caret's twoClassSummary expects the event class as the first factor level.
train_cv <- train_data %>%
  mutate(diabetes = relevel(diabetes, ref = "pos"))

cv_model <- train(
  diabetes ~ pregnancies + glucose + pressure + triceps + insulin +
    mass + pedigree + age,
  data = train_cv,
  method = "glm",
  family = binomial,
  metric = "ROC",
  trControl = ctrl
)
cv_model
cv_model$results

# 11. Test-set predictions
# Refit model with original factor ordering for clear probability/prediction labels.
prob <- predict(logit_model, newdata = test_data, type = "response")
pred <- factor(ifelse(prob >= 0.50, "pos", "neg"),
               levels = c("neg", "pos"))

conf_mat <- confusionMatrix(pred, test_data$diabetes, positive = "pos")
conf_mat

# 12. Performance metrics
accuracy <- as.numeric(conf_mat$overall["Accuracy"])
sensitivity <- as.numeric(conf_mat$byClass["Sensitivity"])
specificity <- as.numeric(conf_mat$byClass["Specificity"])
precision <- as.numeric(conf_mat$byClass["Pos Pred Value"])
f1 <- 2 * precision * sensitivity / (precision + sensitivity)

performance <- data.frame(
  Metric = c("Accuracy", "Sensitivity/Recall", "Specificity",
             "Precision", "F1 Score"),
  Value = c(accuracy, sensitivity, specificity, precision, f1)
)
performance

# 13. ROC-AUC
roc_obj <- roc(
  response = test_data$diabetes,
  predictor = prob,
  levels = c("neg", "pos"),
  direction = "<"
)
auc_value <- as.numeric(auc(roc_obj))
auc_value
plot(roc_obj, main = paste("ROC Curve - AUC =", round(auc_value, 3)))

# 14. Diagnostic analysis
# Deviance residuals and fitted values
diagnostic_df <- data.frame(
  fitted = fitted(logit_model),
  residuals = residuals(logit_model, type = "deviance")
)

ggplot(diagnostic_df, aes(x = fitted, y = residuals)) +
  geom_point() +
  geom_hline(yintercept = 0, linetype = "dashed") +
  labs(title = "Logistic Regression Residuals",
       x = "Fitted Probability", y = "Deviance Residual")

# 15. Save cleaned dataset and model outputs
write.csv(diabetes_clean, "diabetes_cleaned.csv", row.names = FALSE)
write.csv(performance, "week3_model_performance.csv", row.names = FALSE)
